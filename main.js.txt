// Pricing calculator data
const pricingData = {
    poster: { basic: 599, standard: 999, premium: 1499 },
    banner: { basic: 699, standard: 1199, premium: 1699 },
    logo: { basic: 799, standard: 1399, premium: 1999 },

    social: { basic: 599, standard: 999, premium: 1499 }
};

const deliveryMultipliers = {
    normal: 1,
    fast: 1.1,
    express: 1.2
};

// DOM elements
let navbar, mobileMenuBtn, mobileMenu;
let designTypeSelect, complexitySelect, deliverySelect;
let priceResult, calculatedPrice, whatsappCTA;
let portfolioGrid, filterButtons;
let contactForm;

// Initialize when DOM is loaded
document.addEventListener('DOMContentLoaded', function() {
    initializeElements();
    initializeEventListeners();
    initializeAnimations();
    initializePortfolio();
    initializeCalculator();
    initializeScrollEffects();
});

function initializeElements() {
    navbar = document.getElementById('navbar');
    mobileMenuBtn = document.getElementById('mobile-menu-btn');
    mobileMenu = document.getElementById('mobile-menu');
    
    designTypeSelect = document.getElementById('designType');
    complexitySelect = document.getElementById('complexity');
    deliverySelect = document.getElementById('delivery');
    priceResult = document.getElementById('priceResult');
    calculatedPrice = document.getElementById('calculatedPrice');
    whatsappCTA = document.getElementById('whatsappCTA');
    
    portfolioGrid = document.getElementById('portfolio-grid');
    filterButtons = document.querySelectorAll('.filter-btn');
    
    contactForm = document.getElementById('contactForm');
}

function initializeEventListeners() {
    // Mobile menu toggle
    mobileMenuBtn.addEventListener('click', toggleMobileMenu);
    
    // Calculator inputs
    designTypeSelect.addEventListener('change', calculatePrice);
    complexitySelect.addEventListener('change', calculatePrice);
    deliverySelect.addEventListener('change', calculatePrice);
    
    // Portfolio filters
    filterButtons.forEach(btn => {
        btn.addEventListener('click', handleFilterClick);
    });
    
    // Contact form
    contactForm.addEventListener('submit', handleContactForm);
    
    // Smooth scrolling for navigation links
    document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function (e) {
            e.preventDefault();
            const target = document.querySelector(this.getAttribute('href'));
            if (target) {
                target.scrollIntoView({
                    behavior: 'smooth',
                    block: 'start'
                });
            }
        });
    });
    
    // Navbar scroll effect
    window.addEventListener('scroll', handleNavbarScroll);
}

        ],

    // Splitting.js for text animations
    Splitting();
}

function initializePortfolio() {
    renderPortfolio(portfolioData);
}

function initializeCalculator() {
    calculatePrice();
}

function initializeScrollEffects() {
    // Intersection Observer for scroll animations
    const observerOptions = {
        threshold: 0.1,
        rootMargin: '0px 0px -50px 0px'
    };
    
    const observer = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                entry.target.classList.add('revealed');
            }
        });
    }, observerOptions);
    
    // Observe all sections with reveal animation
    document.querySelectorAll('.section-reveal').forEach(el => {
        observer.observe(el);
    });
}

function toggleMobileMenu() {
    mobileMenu.classList.toggle('hidden');
}

function handleNavbarScroll() {
    if (window.scrollY > 50) {
        navbar.classList.add('nav-sticky');
    } else {
        navbar.classList.remove('nav-sticky');
    }
}

function calculatePrice() {
    const designType = designTypeSelect.value;
    const complexity = complexitySelect.value;
    const delivery = deliverySelect.value;
    
    if (!designType || !complexity || !delivery) return;
    
    const basePrice = pricingData[designType][complexity];
    const finalPrice = Math.round(basePrice * deliveryMultipliers[delivery]);
    
    // Animate price display
    anime({
        targets: calculatedPrice,
        innerHTML: [0, finalPrice],
        round: 1,
        duration: 1000,
        easing: 'easeOutQuart',
        update: function(anim) {
            calculatedPrice.innerHTML = `₹${Math.round(anim.animatables[0].target.innerHTML)}`;
        }
    });
    
    // Show results with animation
    priceResult.classList.remove('hidden');
    whatsappCTA.classList.remove('hidden');
    
    anime({
        targets: [priceResult, whatsappCTA],
        opacity: [0, 1],
        translateY: [20, 0],
        duration: 600,
        delay: anime.stagger(200),
        easing: 'easeOutQuart'
    });
    
    // Update WhatsApp link with pre-filled message
    const whatsappLink = whatsappCTA.querySelector('a');
    const message = `Hi Sumit, I'm interested in your ${designType} design services (${complexity} package). The estimated price is ₹${finalPrice}. Can we discuss this further?`;
    whatsappLink.href = `https://wa.me/919999999999?text=${encodeURIComponent(message)}`;
}

function scrollToCalculator() {
    document.getElementById('calculator').scrollIntoView({
        behavior: 'smooth',
        block: 'start'
    });
}

function handleFilterClick(e) {
    const filter = e.target.dataset.filter;
    
    // Update active button
    filterButtons.forEach(btn => {
        btn.classList.remove('active', 'bg-amber-500', 'text-white');
        btn.classList.add('bg-white', 'text-gray-700');
    });
    
    e.target.classList.add('active', 'bg-amber-500', 'text-white');
    e.target.classList.remove('bg-white', 'text-gray-700');
    
    // Filter portfolio items
    const filteredData = filter === 'all' 
        ? portfolioData 
        : portfolioData.filter(item => item.category === filter);
    
    renderPortfolio(filteredData);
}

function renderPortfolio(data) {
    // Clear existing items
    portfolioGrid.innerHTML = '';
    
    // Create portfolio items
    data.forEach((item, index) => {
        const portfolioItem = createPortfolioItem(item);
        portfolioGrid.appendChild(portfolioItem);
        
        // Animate item entrance
        anime({
            targets: portfolioItem,
            opacity: [0, 1],
            translateY: [30, 0],
            duration: 600,
            delay: index * 100,
            easing: 'easeOutQuart'
        });
    });
}

function createPortfolioItem(item) {
    const div = document.createElement('div');
    div.className = 'portfolio-item card-hover cursor-pointer';
    div.innerHTML = `
        <img src="${item.image}" alt="${item.title}" class="w-full h-64 object-cover">
        <div class="overlay-content">
            <h3 class="text-xl font-bold mb-2">${item.title}</h3>
            <p class="text-sm opacity-90">${item.description}</p>
            <button class="mt-4 bg-white text-gray-900 px-4 py-2 rounded-lg font-semibold hover:bg-gray-100 transition-colors">
                View Details
            </button>
        </div>
    `;
    
    // Add click handler for lightbox
    div.addEventListener('click', () => openLightbox(item));
    
    return div;
}

function handleContactForm(e) {
    e.preventDefault();
    
    const formData = {
        name: document.getElementById('name').value,
        email: document.getElementById('email').value,
        projectType: document.getElementById('projectType').value,
        message: document.getElementById('message').value
    };
    
    // Simulate form submission
    const submitBtn = e.target.querySelector('button[type="submit"]');
    const originalText = submitBtn.textContent;
    
    submitBtn.textContent = 'Sending...';
    submitBtn.disabled = true;
    
    setTimeout(() => {
        // Show success message
        showNotification('Message sent successfully! Sumit will get back to you soon.', 'success');
        
        // Reset form
        contactForm.reset();
        
        // Reset button
        submitBtn.textContent = originalText;
        submitBtn.disabled = false;
    }, 2000);
}

// Utility functions
function debounce(func, wait) {
    let timeout;
    return function executedFunction(...args) {
        const later = () => {
            clearTimeout(timeout);
            func(...args);
        };
        clearTimeout(timeout);
        timeout = setTimeout(later, wait);
    };
}
